#!/bin/bash
# Build the task-prep release candidate as ONE patch on top of a GIVEN origin/main commit, in a FRESH tree.
# Usage: build_release.sh <repo with both commits> <candidate sha> <main sha> <new empty dir>
set -euo pipefail
REPO=$1; C=$2; M=$3; REL=$4
PRE=22fe8c8cf21df82c237211330986bd2c2abc2a8c     # integ state before the record-prep package (its mission_control base)
test ! -e "$REL"
git -C "$REPO" worktree add --detach -q "$REL" "$M"
cd "$REL"
NEW="spa_core/studio_os/memory/arch_review.py spa_core/studio_os/memory/role_pack.py
     spa_core/studio_os/model_roles/__init__.py spa_core/studio_os/model_roles/__main__.py
     spa_core/studio_os/model_roles/adapters.py spa_core/studio_os/model_roles/contract.py
     spa_core/studio_os/model_roles/probe.py architecture/model_roles.json"
SAME_BASE="spa_core/studio_os/owner_view.py spa_core/studio_os/mission_ui/app.js spa_core/studio_os/mission_ui/i18n.js
           spa_core/studio_os/mission_ui/styles.css spa_core/studio_os/memory/__main__.py"
TESTS="spa_core/tests/test_task_prep.py spa_core/tests/test_task_prep_record.py spa_core/tests/test_model_roles_contract.py
       spa_core/tests/test_model_roles_review.py spa_core/tests/test_model_roles_probe.py
       spa_core/tests/test_role_pack.py"
DOCS="docs/evidence/studio-os-next-01-2026-10-16/HANDOVER_2026-10-23_STAGE8.md
      docs/evidence/studio-os-next-01-2026-10-16/HANDOVER_2026-10-23_POST_ARB8.md"   # STAGE8: read by test_role_pack; POST_ARB8: --handover of prepare-record
for f in $NEW $TESTS $DOCS; do
  git cat-file -e "${M}:$f" 2>/dev/null && { echo "REFUSED: $f already exists on main — would overwrite"; exit 3; }
  mkdir -p "$(dirname "$f")"; git show "${C}:$f" > "$f"
done
for f in $SAME_BASE; do
  case $f in
    */memory/__main__.py) git diff "${M}:$f" "${C}:$f" | grep '^-[^-]' | grep -qv '^-"""' && { echo "REFUSED: $f main has lines the candidate removes"; exit 4; } ;;
    *) [ "$(git rev-parse "${M}:$f")" = "$(git rev-parse "${PRE}:$f")" ] || { echo "REFUSED: $f changed on main since the package base"; exit 4; } ;;
  esac
  git show "${C}:$f" > "$f"
done
git diff "$PRE" "$C" -- spa_core/studio_os/mission_control.py | git apply --check && \
  git diff "$PRE" "$C" -- spa_core/studio_os/mission_control.py | git apply
git status --short
